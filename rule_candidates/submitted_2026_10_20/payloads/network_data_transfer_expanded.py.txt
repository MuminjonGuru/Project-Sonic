class Severity(Enum):

    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"



class Category(Enum):

@dataclass
class Finding:
    id: str
    title: str
    severity: Severity
    category: Category
    description: str
    evidence: Dict[str, Any]

# AST SCANNER (EXPANDED)
# — — — — — — — — — — — — — — — — — — — — — — — — — — — — — — —

def scan_ast_file(path: Path, context: ScannerContext) -> List[Finding]:
    """Scan Python files for network/data transfer patterns (expanded)."""
    findings = []
    try:
        source = path.read_text(encoding="utf-8")
        tree = ast.parse(source, filename=str(path))

        call_nodes = [n for n in ast.walk(tree) if isinstance(n, ast.Call)]

        for node in call_nodes:
            func_name = _get_call_name(node.func)
            if not func_name:
                continue

            func_lower = func_name.lower()
            all_patterns = SERVICE_PATTERNS.keys()

            if any(f in func_lower for f in [
                "http", "request", "stream", "open", "connect", "grpc", "websockets", "s3", "bucket"
            ]):
                source_snippet = _get_line_snippet(source, node.lineno)

                for rule in context.rules:
                    for pattern in rule["patterns"]:
                        if re.search(pattern, source_snippet, re.IGNORECASE):
                            evidence = rule["evidence_extractor"](node, None) | {
                                "file": str(path),
                                "code_line": source_snippet.strip(),
                                "service": _detect_service(func_lower),
                            }

                            severity = rule["severity"]
                            if _is_known_good_destination(source_snippet):
                                severity = Severity.LOW

                            # Add service-specific tags
                            tags = rule.get("tags", [])
                            if "stream" in func_lower:
                                tags.append("streaming")
                            if "grpc" in func_lower:
                                tags.append("grpc")

                            findings.append(
                                Finding(
                                    id=rule["id"],
                                    title=rule["name"],
                                    severity=severity,
                                    category=rule["category"],
                                    description=f"Found {func_name} usage matching pattern: {pattern}",
                                    evidence=evidence,
                                    location=f"{path}:{node.lineno}",
                                    rule_id=rule["id"],
                                    remediation_hint=rule.get("remediation_hint"),
                                    references=[
                                        "https://owasp.org/www-community/attacks/Server-Side_Request_Forgery",
                                        "https://cheatsheetseries.owasp.org/cheatsheets/SSRF_Bypass_Cheat_Sheet.html",
                                    ],
                                    tags=tags,
                                    context={
                                        "framework": "generic",
                                        "service_type": _detect_service(func_lower),
                                        "streaming": "stream" in func_lower,
                                    },
                                    file_line_range=(node.lineno, node.end_lineno or node.lineno),
                                )
                            )
    except Exception as e:
        findings.append(
            Finding(
                id="NET-DATA-099",
                title="AST parse failed",
                severity=Severity.LOW,
                category=Category.SIDE_CHANNEL,
                description=f"Failed to parse {path}: {e}",
                evidence={"exception": str(e)},
                context={"path": str(path)},
            )
        )
    return findings



def _get_call_name(func_node: ast.AST) -> Optional[str]:
    if isinstance(func_node, ast.Name):
        return func_node.id
    if isinstance(func_node, ast.Attribute):
        return f"{_get_call_name(func_node.value)}.{func_node.attr}"
    if isinstance(func_node, ast.Call):
        return f"{_get_call_name(func_node.func)}()"
    return None



def _get_line_snippet(source: str, lineno: int, context_lines: int = 3) -> str:
    lines = source.splitlines()
    start = max(0, lineno - 1 - context_lines)
    end = min(len(lines), lineno + context_lines)
    return "\n".join(
        f"{i + 1:>5} | {line}"
        for i, line in enumerate(lines[start:end], start=start)
    )



def _is_known_good_destination(snippet: str) -> bool:
    url_match = re.search(r"https?://[^\s,')\"\]]+", snippet)
    if not url_match:
        return False
    host = url_match.group(0).replace("http://", "").replace("https://", "").split("/")[0]
    return any(host.endswith(k) for k in KNOWN_GOOD_HOSTS)



def _detect_service(func_lower: str) -> str:
    if "grpc" in func_lower:
        return "grpc"
    if "websockets" in func_lower or "ws://" in func_lower:
        return "websocket"
    if "s3" in func_lower or "boto" in func_lower:
        return "s3"
    return "http"

def trace_credentials(context: ScannerContext) -> Dict[str, Path]:
    """Build a map of credential names to file locations where they’re defined."""
    cred_map: Dict[str, Path] = {}
    for py_file in context.source_path.rglob("*.py"):
        try:
            source = py_file.read_text()
            tree = ast.parse(source, filename=str(py_file))
            for node in ast.walk(tree):
                if isinstance(node, ast.Assign):
                    for target in node.targets:
                        if isinstance(target, ast.Name):
                            name = target.id.lower()
                            if any(k in name for k in CREDENTIAL_HINTS.keys()):
                                for cred_type, hints in CREDENTIAL_HINTS.items():
                                    if any(h in name for h in hints):
                                        cred_map[target.id] = py_file
        except Exception:
            continue
    return cred_map



def trace_flow_from_secret(secret_name: str, context: ScannerContext) -> List[Dict[str, Any]]:
    """Trace usage of a secret across code (URLs, headers, etc.)."""
    flows = []
    for py_file in context.source_path.rglob("*.py"):
        try:
            source = py_file.read_text()
            if secret_name in source:
                # Extract context lines
                lines = source.splitlines()
                for i, line in enumerate(lines, start=1):
                    if secret_name in line and any(f in line for f in ["http", "header", "auth", "url"]):
                        flows.append({
                            "file": str(py_file),
                            "line": i,
                            "context": line.strip(),
                            "usage": "url" if "http" in line else "header" if "auth" in line else "other",
                        })
        except Exception:
            continue
    return flows

def scan_mcp_configs(root: Path, context: ScannerContext) -> List[Finding]:
    """Scan MCP configs for network/data transfer patterns (expanded)."""
    findings = []
    config_files = list(root.rglob("mcp*.yaml")) + list(root.rglob("mcp*.yml")) + list(root.rglob("mcp*.json"))
    for cfg_path in config_files:
        try:
            text = cfg_path.read_text()
            data = _load_config(text, cfg_path.suffix)
            for tool in data.get("tools", []) if isinstance(data, dict) else []:
                name = tool.get("name", "unnamed")
                config_str = str(tool)
                if any(p in config_str.lower() for p in ["url", "endpoint", "host", "stream"]):
                    severity = Severity.HIGH if "stream" in config_str.lower() else Severity.MEDIUM
                    findings.append(
                        Finding(
                            id="NET-DATA-MCP-001",
                            title=f"Tool '{name}' may perform network/data transfers",
                            severity=severity,
                            category=Category.DYNAMIC_DESTINATION,
                            description=f"MCP config defines network-capable tool.",
                            evidence={
                                "file": str(cfg_path),
                                "tool_name": name,
                                "tool_schema": tool,
                                "field_match": [
                                    k for k in tool.keys() if k.lower() in ["url", "endpoint", "host", "stream"]
                                ],
                            },
                            location=f"{cfg_path}:{cfg_path.read_text().find(name)}",
                            rule_id="NET-DATA-MCP-001",
                            remediation_hint="Pin endpoints and use secrets for dynamic destinations.",
                            references=["https://github.com/NVIDIA/SkillSpector"],
                            tags=["MCP", "network", "stream" if "stream" in config_str.lower() else "request"],
                            context={"config_type": "yaml" if cfg_path.suffix == ".yaml" else "json"},
                        )
                    )
        except Exception as e:
            findings.append(
                Finding(
                    id="NET-DATA-MCP-999",
                    title=f"Failed to parse MCP config: {cfg_path}",
                    severity=Severity.LOW,
                    category=Category.SIDE_CHANNEL,
                    description=str(e),
                    evidence={"path": str(cfg_path)},
                )
            )
    return findings



def _load_config(text: str, ext: str) -> Dict[str, Any]:
    if ext in (".json",):
        return json.loads(text)
    import yaml
    return yaml.safe_load(text) or {}

def run_network_data_scan(
    source_path: Path,
    context: Optional[ScannerContext] = None,
    scan_mcp: bool = True,
    min_severity: Severity = Severity.LOW,
    trace_credentials_flag: bool = True,
) -> List[Finding]:
    """Main entrypoint for network & data transfer analysis (expanded)."""
    ctx = context or ScannerContext(
        source_path=source_path,
        rules=DEFAULT_RULES["network_data_transfer"]["rules"].copy(),
    )

    # Trace credentials if enabled
    if trace_credentials_flag:
        ctx.credential_map = trace_credentials(ctx)

    all_findings: List[Finding] = []

    py_files = list(source_path.rglob("*.py"))
    for py_file in py_files:
        all_findings.extend(scan_ast_file(py_file, ctx))

    if scan_mcp:
        all_findings.extend(scan_mcp_configs(source_path, ctx))

    # Deduplicate
    seen = set()
    unique_findings: List[Finding] = []
    for f in all_findings:
        key = (f.id, f.location, hash(tuple(sorted(f.evidence.items()))))
        if key not in seen:
            seen.add(key)
            unique_findings.append(f)

    # Sort by severity
    unique_findings.sort(key=lambda f: f.severity.value, reverse=True)

    return unique_findings
