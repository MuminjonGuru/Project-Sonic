class Severity(Enum):
    CRITICAL = "critical"
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"



class Category(Enum):
    SELF_PROMPT = "self_prompt"
    TOOL_REGENERATION = "tool_regeneration"
    MEMORY_STRATEGY = "memory_strategy"
    CONTEXT_TUNING = "context_tuning"

    VALIDATION = "validation"



@dataclass
class Finding:
    id: str
    title: str
    severity: Severity
    category: Category
    description: str
    evidence: Dict[str, Any]

def is_self_modification(func_name: str) -> bool:
    """Detect if function is likely a self-modification operation."""
    func_lower = func_name.lower()
    return any(
        p in func_lower
        for p in ["update", "rewrite", "regenerate", "tune", "evaluate", "self"]
    )



def analyze_self_mod_persistence(path: str, content: Optional[str] = None) -> Dict[str, Any]:
    """Analyze self-modification persistence: file, db, or cloud."""
    if content and any(p in content for p in ["db.save", "redis.set", "sqlite.execute"]):
        return {"type": "db", "confidence": 0.8}
    if any(path.endswith(ext) for ext in [".json", ".yaml", ".pkl"]):
        return {"type": "file", "confidence": 0.7}
    if any(ext in path.lower() for ext in ["/s3/", "/gcs/", "azure/"]):
        return {"type": "cloud", "confidence": 0.9}
    return {"type": "unknown", "confidence": 0.5}

def is_llm_driven_modification(code_snippet: str) -> Dict[str, Any]:
    """Detect if modification is LLM-driven."""
    llm_indicators = 0
    if re.search(r"llm\.(decide|suggest|generate|propose)", code_snippet):
        llm_indicators += 25
    if re.search(r"prompt\s*=\s*llm", code_snippet):
        llm_indicators += 20
    if re.search(r"llm\.suggest.*self", code_snippet):
        llm_indicators += 25
    if re.search(r"agent\.response\s*\.\s*(update|rewrite)", code_snippet):
        llm_indicators += 30

    return {
        "llm_driven_score": min(100, llm_indicators),
        "llm_drivers": [
            "decide",
            "suggest",
            "generate",
            "propose",
        ],
        "is_llm_driven": llm_indicators >= 40,
    }

def detect_framework(path: Path, source: str) -> List[str]:
    """Detect which agent frameworks are used in the code."""
    frameworks = []
    for fw, patterns in FRAMEWORK_PATTERNS.items():
        if any(re.search(p, source, re.IGNORECASE) for p in patterns):
            frameworks.append(fw)
    return frameworks

# === SECURITY SCORING ENGINE (EXPANDED) ===

def compute_risk_score(finding: Finding, context: ScannerContext = None) -> Dict[str, Any]:
    """Compute dynamic risk score (0–100) and confidence level for self-modification findings."""
    base = {
        "critical": 90,
        "high": 70,
        "medium": 50,
        "low": 30,
    }.get(finding.severity.value, 40)

    evidence = finding.evidence
    llm_driven = evidence.get("llm_driven", {})
    injection = evidence.get("injection", {})
    validation = evidence.get("validation", {})
    drift = evidence.get("drift", {})
    loops = evidence.get("loops", {})
    overflow = evidence.get("overflow", {})
    to_code = evidence.get("to_code", {})
    persistence = evidence.get("persistence", {})
    framework_mismatch = evidence.get("framework_mismatch", {})

    llm_modifier = llm_driven.get("llm_driven_score", 0) if isinstance(llm_driven, dict) else 0
    injection_modifier = injection.get("injection_risk_score", 0) if isinstance(injection, dict) else 0
    validation_modifier = validation.get("validation_risk_score", 0) if isinstance(validation, dict) else 0
    drift_modifier = drift.get("drift_risk_score", 0) if isinstance(drift, dict) else 0
    loops_modifier = loops.get("loops_risk_score", 0) if isinstance(loops, dict) else 0
    overflow_modifier = overflow.get("overflow_risk_score", 0) if isinstance(overflow, dict) else 0
    to_code_modifier = to_code.get("to_code_risk_score", 0) if isinstance(to_code, dict) else 0
    persistence_modifier = persistence.get("persistence_risk_score", 0) if isinstance(persistence, dict) else 0
    framework_modifier = framework_mismatch.get("mismatch_risk_score", 0) if isinstance(framework_mismatch, dict) else 0

    final_score = min(
        100,
        base
        + llm_modifier
        + injection_modifier
        + validation_modifier
        + drift_modifier
        + loops_modifier
        + overflow_modifier
        + to_code_modifier
        + persistence_modifier
        + framework_modifier,
    )

    evidence_items = list(evidence.items())
    confidence = 0.5 + (len(evidence_items) / 12)

    return {
        "risk_score": final_score,
        "llm_modifier": llm_modifier,
        "injection_modifier": injection_modifier,
        "validation_modifier": validation_modifier,
        "drift_modifier": drift_modifier,
        "loops_modifier": loops_modifier,
        "overflow_modifier": overflow_modifier,
        "to_code_modifier": to_code_modifier,
        "persistence_modifier": persistence_modifier,
        "framework_modifier": framework_modifier,
        "confidence": round(confidence, 2),
    }

# === AST SCANNER (EXPANDED) — PART 1 — ===

def scan_ast_file(path: Path, context: ScannerContext) -> List[Finding]:
    """Scan Python files for agent self-modification patterns (expanded)."""
    findings = []
    try:
        source = path.read_text(encoding="utf-8")
        tree = ast.parse(source, filename=str(path))

        call_nodes = [n for n in ast.walk(tree) if isinstance(n, ast.Call)]

        for node in call_nodes:
            func_name = _get_call_name(node.func)
            if not func_name:
                continue

            func_lower = func_name.lower()
            if any(f in func_lower for f in ["update", "rewrite", "regenerate", "tune", "evaluate", "self"]):
                source_snippet = _get_line_snippet(source, node.lineno)

                for rule in context.rules:
                    for pattern in rule["patterns"]:
                        if re.search(pattern, source_snippet, re.IGNORECASE):
                            # Compute additional metrics
                            llm_driven = is_llm_driven_modification(source_snippet)
                            injection = score_self_mod_injection(source_snippet)
                            validation = score_validation_hooks(source_snippet)
                            drift = score_self_mod_drift(source_snippet)
                            loops = score_self_mod_loops(source_snippet)
                            overflow = score_context_overflow_self_mod(source_snippet)
                            to_code = score_memory_to_code_self_mod(source_snippet)
                            persistence = score_persistence(source_snippet)
                            framework = detect_framework(path, source_snippet)
                            framework_mismatch = score_framework_mismatch(source_snippet, framework)

                            evidence = rule["evidence_extractor"](node, None) | {
                                "file": str(path),
                                "code_line": source_snippet.strip(),
                                "llm_driven": llm_driven,
                                "injection": injection,
                                "validation": validation,
                                "drift": drift,
                                "loops": loops,
                                "overflow": overflow,
                                "to_code": to_code,
                                "persistence": persistence,
                                "framework_mismatch": framework_mismatch,
                                "frameworks": framework,
                            }

                            findings.append(
                                Finding(
                                    id=rule["id"],
                                    title=rule["name"],
                                    severity=rule["severity"],
                                    category=rule["category"],
                                    description=f"Found {func_name} usage matching pattern: {pattern}",
                                    evidence=evidence,
                                    location=f"{path}:{node.lineno}",
                                    rule_id=rule["id"],
                                    remediation_hint=rule.get("remediation_hint"),
                                    references=[
                                        "https://langchain-ai.github.io/langchain/",
                                        "https://microsoft.github.io/autogen/",
                                    ],
                                    tags=[
                                        "self-update" if "update" in func_lower else "self-mod",
                                        "llm-driven" if llm_driven["is_llm_driven"] else None,
                                        "injection" if injection["is_vulnerable"] else None,
                                        "validation" if validation["is_thorough"] else None,
                                        "persistence" if "save" in func_lower else None,
                                    ] + framework,
                                    context={
                                        "framework": framework[0] if framework else "generic",
                                        "llm_driven": llm_driven,
                                        "injection": injection,
                                        "validation": validation,
                                        "drift": drift,
                                        "loops": loops,
                                        "overflow": overflow,
                                        "to_code": to_code,
                                        "persistence": persistence,
                                        "framework_mismatch": framework_mismatch,
                                    },
                                    file_line_range=(node.lineno, node.end_lineno or node.lineno),
                                )
                            )
    except Exception as e:
        findings.append(
            Finding(
                id="SELF-MOD-099",
                title="AST parse failed",
                severity=Severity.LOW,
                category=Category.SELF_PROMPT,
                description=f"Failed to parse {path}: {e}",
                evidence={"exception": str(e)},
                context={"path": str(path)},
            )
        )
    return findings



def _get_call_name(func_node: ast.AST) -> Optional[str]:
    if isinstance(func_node, ast.Name):
        return func_node.id
    if isinstance(func_node, ast.Attribute):
        return f"{_get_call_name(func_node.value)}.{func_node.attr}"
    if isinstance(func_node, ast.Call):
        return f"{_get_call_name(func_node.func)}()"
    return None



def _get_line_snippet(source: str, lineno: int, context_lines: int = 3) -> str:
    lines = source.splitlines()
    start = max(0, lineno - 1 - context_lines)
    end = min(len(lines), lineno + context_lines)
    return "\n".join(
        f"{i + 1:>5} | {line}"
        for i, line in enumerate(lines[start:end], start=start)
    )
# === MCP SELF-MODIFICATION STORAGE PARSER (PART 2) ===

def parse_mcp_self_mod_storage(root: Path) -> List[Dict[str, Any]]:
    """Parse MCP config for self-modification storage patterns (e.g., state stores, context stores)."""
    state_stores = []
    config_files = list(root.rglob("mcp*.yaml")) + list(root.rglob("mcp*.yml")) + list(root.rglob("mcp*.json"))
    for cfg_path in config_files:
        try:
            text = cfg_path.read_text()
            data = json.loads(text) if cfg_path.suffix == ".json" else _load_yaml(text)

            # Look for self-mod storage at top-level or under 'state' / 'memory' keys
            if "state" in data:
                states = data["state"] if isinstance(data["state"], list) else [data["state"]]
                for state in states:
                    if isinstance(state, dict):
                        if "self" in str(state.get("path", "")).lower():
                            state_stores.append({
                                "path": state.get("path", "/"),
                                "type": state.get("type", "memory"),
                                "mode": state.get("mode", "read-write"),
                                "source": str(cfg_path),
                            })
            if "memory" in data:
                memories = data["memory"] if isinstance(data["memory"], list) else [data["memory"]]
                for mem in memories:
                    if isinstance(mem, dict):
                        if "self" in str(mem.get("path", "")).lower():
                            state_stores.append({
                                "path": mem.get("path", "/"),
                                "type": mem.get("type", "chat"),
                                "mode": mem.get("mode", "read-write"),
                                "source": str(cfg_path),
                            })
        except Exception:
            continue
    return state_stores



def _load_yaml(text: str) -> Dict[str, Any]:
    import yaml
    return yaml.safe_load(text) or {}

                print(f" Validation: {f.evidence['validation']['validation_risk_score']}/100")
            if f.evidence.get("drift"):
                print(f" Drift: {f.evidence['drift']['drift_risk_score']}/100")
            if f.evidence.get("loops"):
                print(f" Loops: {f.evidence['loops']['loops_risk_score']}/100")
            if f.evidence.get("overflow"):
                print(f" Context Overflow: {f.evidence['overflow']['overflow_risk_score']}/100")
            if f.evidence.get("to_code"):
                print(f" Memory-to-Code: {f.evidence['to_code']['to_code_risk_score']}/100")
            if f.evidence.get("framework_mismatch"):
                print(f" Framework Mismatch: {f.evidence['framework_mismatch']['mismatch_risk_score']}/100")
            if f.remediation_hint:
                print(f" Remediation: {f.remediation_hint}")

        if args.ci:
            critical_count = sum(1 for f in unique_findings if f.severity == Severity.CRITICAL)
            if critical_count > 0:
                print(f"\n❌ CI fail: {critical_count} critical findings")
                exit(1)
            else:
                print(f"\n✅ CI pass: {len(unique_findings)} findings")
