def is_orchestration_operation(func_name: str) -> bool:
    """Detect if function is likely an orchestration operation."""
    func_lower = func_name.lower()
    return any(
        p in func_lower
        for p in ["orchestrat", "route", "handoff", "forward", "dispatch", "share"]
    )



def analyze_orchestration_persistence(path: str, content: Optional[str] = None) -> Dict[str, Any]:
    """Analyze orchestration persistence: file, db, or cloud."""
    if content and any(p in content for p in ["db.save", "redis.set", "sqlite.execute"]):
        return {"type": "db", "confidence": 0.8}
    if any(path.endswith(ext) for ext in [".json", ".yaml", ".pkl"]):
        return {"type": "file", "confidence": 0.7}
    if any(ext in path.lower() for ext in ["/s3/", "/gcs/", "azure/"]):
        return {"type": "cloud", "confidence": 0.9}
    return {"type": "unknown", "confidence": 0.5}



# === LLM-DRIVEN DETECTION (ENHANCED) ===

def is_llm_driven_orchestration(code_snippet: str) -> Dict[str, Any]:
    """Detect if orchestration is LLM-driven."""
    llm_indicators = 0
    if re.search(r"orchestrator\.(route|select|dispatch).*llm", code_snippet):
        llm_indicators += 30
    if re.search(r"llm\.suggest.*orchestrat|llm\.propose.*orchestrat", code_snippet):
        llm_indicators += 25
    if re.search(r"orchestrator\.response\s*\.\s*(route|update)", code_snippet):
        llm_indicators += 25
    if re.search(r"agent\.response\s*\.\s*orchestrat", code_snippet):
        llm_indicators += 20

    return {
        "llm_driven_score": min(100, llm_indicators),
        "llm_drivers": [
            "route",
            "select",
            "dispatch",
            "propose",
        ],
        "is_llm_driven": llm_indicators >= 50,
    }

def detect_framework(path: Path, source: str) -> List[str]:
    """Detect which agent frameworks are used in the code."""
    frameworks = []
    for fw, patterns in FRAMEWORK_PATTERNS.items():
        if any(re.search(p, source, re.IGNORECASE) for p in patterns):
            frameworks.append(fw)
    return frameworks



def score_framework_mismatch(code_snippet: str, frameworks: List[str]) -> Dict[str, Any]:
    """Score how likely framework-specific patterns are mismatched."""
    mismatch_risks = 0
    if "langchain" in frameworks and re.search(r"AgentExecutor", code_snippet):
        mismatch_risks += 20
    if "autogen" in frameworks and re.search(r"GroupChat", code_snippet):
        mismatch_risks += 20
    if "mcp" in frameworks and re.search(r"orchestrator\.state\.store", code_snippet):
        mismatch_risks += 20
    if len(frameworks) > 2:
        mismatch_risks += 20 # multi-framework = higher mismatch risk

    return {
        "mismatch_risk_score": min(100, mismatch_risks),
        "mismatch_vectors": frameworks,
        "is_vulnerable": mismatch_risks >= 40,
    }

# === SECURITY SCORING ENGINE (ENHANCED) ===

def compute_risk_score(finding: Finding, context: ScannerContext = None) -> Dict[str, Any]:
    """Compute dynamic risk score (0–100) and confidence level for orchestration findings."""
    base = {
        "critical": 90,
        "high": 70,
        "medium": 50,
        "low": 30,
    }.get(finding.severity.value, 40)

    evidence = finding.evidence
    llm_driven = evidence.get("llm_driven", {})
    nested = evidence.get("nested", {})
    hijacking = evidence.get("hijacking", {})
    routing = evidence.get("routing", {})
    shared_state = evidence.get("shared_state", {})
    leakage = evidence.get("leakage", {})
    framework_mismatch = evidence.get("framework_mismatch", {})
    self_mod = evidence.get("self_mod", {})

    llm_modifier = llm_driven.get("llm_driven_score", 0) if isinstance(llm_driven, dict) else 0
    nested_modifier = nested.get("nested_risk_score", 0) if isinstance(nested, dict) else 0
    hijacking_modifier = hijacking.get("hijack_risk_score", 0) if isinstance(hijacking, dict) else 0
    routing_modifier = routing.get("routing_risk_score", 0) if isinstance(routing, dict) else 0
    shared_state_modifier = shared_state.get("injection_risk_score", 0) if isinstance(shared_state, dict) else 0
    leakage_modifier = leakage.get("leakage_risk_score", 0) if isinstance(leakage, dict) else 0
    framework_modifier = framework_mismatch.get("mismatch_risk_score", 0) if isinstance(framework_mismatch, dict) else 0
    self_mod_modifier = self_mod.get("self_mod_risk_score", 0) if isinstance(self_mod, dict) else 0

    final_score = min(
        100,
        base
        + llm_modifier
        + nested_modifier
        + hijacking_modifier
        + routing_modifier
        + shared_state_modifier
        + leakage_modifier
        + framework_modifier
        + self_mod_modifier,
    )

    evidence_items = list(evidence.items())
    confidence = 0.5 + (len(evidence_items) / 12)

    return {
        "risk_score": final_score,
        "llm_modifier": llm_modifier,
        "nested_modifier": nested_modifier,
        "hijacking_modifier": hijacking_modifier,
        "routing_modifier": routing_modifier,
        "shared_state_modifier": shared_state_modifier,
        "leakage_modifier": leakage_modifier,
        "framework_modifier": framework_modifier,
        "self_mod_modifier": self_mod_modifier,
        "confidence": round(confidence, 2),
    }

# === AST SCANNER (ENHANCED) — PART 1 — ===

def scan_ast_file(path: Path, context: ScannerContext) -> List[Finding]:
    """Scan Python files for agent orchestration patterns (enhanced)."""
    findings = []
    try:
        source = path.read_text(encoding="utf-8")
        tree = ast.parse(source, filename=str(path))

        call_nodes = [n for n in ast.walk(tree) if isinstance(n, ast.Call)]

        for node in call_nodes:
            func_name = _get_call_name(node.func)
            if not func_name:
                continue

            func_lower = func_name.lower()
            if any(f in func_lower for f in ["orchestrat", "route", "handoff", "forward", "dispatch", "share"]):
                source_snippet = _get_line_snippet(source, node.lineno)

                for rule in context.rules:
                    for pattern in rule["patterns"]:
                        if re.search(pattern, source_snippet, re.IGNORECASE):
                            # Compute additional metrics
                            llm_driven = is_llm_driven_orchestration(source_snippet)
                            nested = score_nested_orchestration(source_snippet)
                            hijacking = score_handoff_hijacking(source_snippet)
                            routing = score_dynamic_routing(source_snippet)
                            shared_state = score_shared_state_injection(source_snippet)
                            leakage = score_cross_agent_leakage(source_snippet)
                            self_mod = score_orchestrator_self_mod(source_snippet)
                            framework = detect_framework(path, source_snippet)
                            framework_mismatch = score_framework_mismatch(source_snippet, framework)

                            evidence = rule["evidence_extractor"](node, None) | {
                                "file": str(path),
                                "code_line": source_snippet.strip(),
                                "llm_driven": llm_driven,
                                "nested": nested,
                                "hijacking": hijacking,
                                "routing": routing,
                                "shared_state": shared_state,
                                "leakage": leakage,
                                "self_mod": self_mod,
                                "framework_mismatch": framework_mismatch,
                                "frameworks": framework,
                            }

                            findings.append(
                                Finding(
                                    id=rule["id"],
                                    title=rule["name"],
                                    severity=rule["severity"],
                                    category=rule["category"],
                                    description=f"Found {func_name} usage matching pattern: {pattern}",
                                    evidence=evidence,
                                    location=f"{path}:{node.lineno}",
                                    rule_id=rule["id"],
                                    remediation_hint=rule.get("remediation_hint"),
                                    references=[
                                        "https://langchain-ai.github.io/langchain/",
                                        "https://microsoft.github.io/autogen/",
                                    ],
                                    tags=[
                                        "orchestration" if "orchestrat" in func_lower else "routing",
                                        "llm-driven" if llm_driven["is_llm_driven"] else None,
                                        "nested" if nested["is_nested"] else None,
                                        "hijacking" if hijacking["is_vulnerable"] else None,
                                        "routing" if routing["is_dynamic"] else None,
                                    ] + framework,
                                    context={
                                        "framework": framework[0] if framework else "generic",
                                        "llm_driven": llm_driven,
                                        "nested": nested,
                                        "hijacking": hijacking,
                                        "routing": routing,
                                        "shared_state": shared_state,
                                        "leakage": leakage,
                                        "self_mod": self_mod,
                                        "framework_mismatch": framework_mismatch,
                                    },
                                    file_line_range=(node.lineno, node.end_lineno or node.lineno),
                                )
                            )
    except Exception as e:
        findings.append(
            Finding(
                id="ORG-099",
                title="AST parse failed",
                severity=Severity.LOW,
                category=Category.DYNAMIC_ROUTING,
                description=f"Failed to parse {path}: {e}",
                evidence={"exception": str(e)},
                context={"path": str(path)},
            )
        )
    return findings



def _get_call_name(func_node: ast.AST) -> Optional[str]:
    if isinstance(func_node, ast.Name):
        return func_node.id
    if isinstance(func_node, ast.Attribute):
        return f"{_get_call_name(func_node.value)}.{func_node.attr}"
    if isinstance(func_node, ast.Call):
        return f"{_get_call_name(func_node.func)}()"
    return None



def _get_line_snippet(source: str, lineno: int, context_lines: int = 3) -> str:
    lines = source.splitlines()
    start = max(0, lineno - 1 - context_lines)
    end = min(len(lines), lineno + context_lines)
    return "\n".join(
        f"{i + 1:>5} | {line}"
        for i, line in enumerate(lines[start:end], start=start)
    )
# === MCP ORCHESTRATION STORAGE PARSER (PART 2) ===

def parse_mcp_orchestration_storage(root: Path) -> List[Dict[str, Any]]:
    """Parse MCP config for orchestration storage patterns (e.g., state stores, memory stores)."""
    state_stores = []
    config_files = list(root.rglob("mcp*.yaml")) + list(root.rglob("mcp*.yml")) + list(root.rglob("mcp*.json"))
    for cfg_path in config_files:
        try:
            text = cfg_path.read_text()
            data = json.loads(text) if cfg_path.suffix == ".json" else _load_yaml(text)

            # Look for orchestration storage at top-level or under 'orchestrator' / 'state' / 'memory' keys
            if "orchestrator" in data:
                orchestrator = data["orchestrator"]
                if isinstance(orchestrator, dict):
                    if "state" in orchestrator or "memory" in orchestrator:
                        state = orchestrator.get("state", orchestrator.get("memory", {}))
                        if isinstance(state, dict):
                            state_stores.append({
                                "path": state.get("path", "/"),
                                "type": state.get("type", "shared"),
                                "mode": state.get("mode", "read-write"),
                                "source": str(cfg_path),
                            })
            if "state" in data:
                states = data["state"] if isinstance(data["state"], list) else [data["state"]]
                for state in states:
                    if isinstance(state, dict):
                        if "orchestrat" in str(state.get("path", "")).lower():
                            state_stores.append({
                                "path": state.get("path", "/"),
                                "type": state.get("type", "shared"),
                                "mode": state.get("mode", "read-write"),
                                "source": str(cfg_path),
                            })
            if "memory" in data:
                memories = data["memory"] if isinstance(data["memory"], list) else [data["memory"]]
                for mem in memories:
                    if isinstance(mem, dict):
                        if "orchestrat" in str(mem.get("path", "")).lower():
                            state_stores.append({
                                "path": mem.get("path", "/"),
                                "type": mem.get("type", "shared"),
                                "mode": mem.get("mode", "read-write"),
                                "source": str(cfg_path),
                            })
        except Exception:
            continue
    return state_stores



def _load_yaml(text: str) -> Dict[str, Any]:
    import yaml
    return yaml.safe_load(text) or {}
