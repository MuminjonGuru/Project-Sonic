def is_memory_operation(func_name: str) -> bool:
    """Detect if function is likely a memory/state operation."""
    func_lower = func_name.lower()
    return any(
        p in func_lower
        for p in ["save", "load", "write", "persist", "archive", "cache", "memory"]
    )



def analyze_state_persistence(path: str, content: Optional[str] = None) -> Dict[str, Any]:
    """Analyze state persistence: file, db, or cloud."""
    if content and any(p in content for p in ["db.save", "redis.set", "sqlite.execute"]):
        return {"type": "db", "confidence": 0.8}
    if any(path.endswith(ext) for ext in [".json", ".yaml", ".pkl"]):
        return {"type": "file", "confidence": 0.7}
    if any(ext in path.lower() for ext in ["/s3/", "/gcs/", "azure/"]):
        return {"type": "cloud", "confidence": 0.9}
    return {"type": "unknown", "confidence": 0.5}

    "api_keys": ["api_key", "secret_key", "token", "credential"],
    "credentials": ["password", "passwd", "pwd", "password", "passphrase"],
    "secrets": ["secret", "key", "token", "oauth", "jwt"],

def detect_secrets_in_state(content: str) -> List[Dict[str, Any]]:
    """Detect unmasked secrets in state files/memory dumps."""
    secrets_found = []
    for cat, patterns in SECRET_PATTERNS.items():
        for pattern in patterns:
            matches = re.findall(rf"(?i){pattern}[\s:=]+\s*['\"]?[\w\-_]+", content)
            if matches:
                secrets_found.append({
                    "category": cat,
                    "pattern": pattern,
                    "matches": matches[:3],
                })
    return secrets_found



# === STATE TAMPERING SCORER (EXPANDED) ===

def score_state_tampering(path: str, content: Optional[str] = None) -> Dict[str, Any]:
    """Score how vulnerable a state file is to tampering."""
    tamper_risks = 0
    if "write" in path or "save" in path:
        tamper_risks += 20
    if content and ("eval(" in content or "exec(" in content):
        tamper_risks += 30
    if any(ext in path for ext in [".py", ".yaml", ".yml"]):
        tamper_risks += 20
    if not content or ("read_text()" in content and "json.loads()" in content):
        tamper_risks += 10

    return {
        "tamper_risk_score": min(100, tamper_risks),
        "tamper_vectors": [
            "direct-write",
            "code-exec",
            "file-format" if "yaml" in path else None,
        ],
        "is_vulnerable": tamper_risks >= 40,
    }

def detect_framework(path: Path, source: str) -> List[str]:
    """Detect which agent frameworks are used in the code."""
    frameworks = []
    for fw, patterns in FRAMEWORK_PATTERNS.items():
        if any(re.search(p, source, re.IGNORECASE) for p in patterns):
            frameworks.append(fw)
    return frameworks

# === AGENT SELF-MEMORY DETECTOR (NEW) ===

def detect_agent_self_memory(code_snippet: str) -> Dict[str, Any]:
    """Detect agent self-memory writes (e.g., agent.update_memory())."""
    self_mem = {
        "is_self_memory": bool(
            re.search(r"agent\.update_memory|agent\.save_memory|agent\.write_memory", code_snippet)
            or re.search(r"self\.memory\s*=\s*|self\.update_memory", code_snippet)
        ),
        "method": None,
    }

    if re.search(r"agent\.", code_snippet):
        self_mem["method"] = "agent"
    elif re.search(r"self\.", code_snippet):
        self_mem["method"] = "self"
    elif re.search(r"memory\.store|memory\.save", code_snippet):
        self_mem["method"] = "store"

    return self_mem

# === SECURITY SCORING ENGINE (EXPANDED) ===

def compute_risk_score(finding: Finding, context: ScannerContext = None) -> Dict[str, Any]:
    """Compute dynamic risk score (0–100) and confidence level for memory/state findings."""
    base = {
        "critical": 90,
        "high": 70,
        "medium": 50,
        "low": 30,
    }.get(finding.severity.value, 40)

    evidence = finding.evidence
    tampering = evidence.get("tampering", {})
    overflow = evidence.get("context_overflow", {})
    compression = evidence.get("compression", {})
    to_code = evidence.get("to_code", {})
    exfil = evidence.get("exfiltration", {})
    versioning = evidence.get("versioning", {})
    poisoning = evidence.get("poisoning", {})

    tampering_modifier = tampering.get("tamper_risk_score", 0) if isinstance(tampering, dict) else 0
    overflow_modifier = overflow.get("overflow_risk_score", 0) if isinstance(overflow, dict) else 0
    compression_modifier = compression.get("compression_risk_score", 0) if isinstance(compression, dict) else 0
    to_code_modifier = to_code.get("to_code_risk_score", 0) if isinstance(to_code, dict) else 0
    exfil_modifier = exfil.get("exfil_risk_score", 0) if isinstance(exfil, dict) else 0
    versioning_modifier = versioning.get("versioning_risk_score", 0) if isinstance(versioning, dict) else 0
    poisoning_modifier = poisoning.get("poisoning_risk_score", 0) if isinstance(poisoning, dict) else 0

    final_score = min(
        100,
        base
        + tampering_modifier
        + overflow_modifier
        + compression_modifier
        + to_code_modifier
        + exfil_modifier
        + versioning_modifier
        + poisoning_modifier,
    )

    evidence_items = list(evidence.items())
    confidence = 0.5 + (len(evidence_items) / 12)

    return {
        "risk_score": final_score,
        "tampering_modifier": tampering_modifier,
        "overflow_modifier": overflow_modifier,
        "compression_modifier": compression_modifier,
        "to_code_modifier": to_code_modifier,
        "exfil_modifier": exfil_modifier,
        "versioning_modifier": versioning_modifier,
        "poisoning_modifier": poisoning_modifier,
        "confidence": round(confidence, 2),
    }

# === AST SCANNER (EXPANDED) — PART 1 — ===

def scan_ast_file(path: Path, context: ScannerContext) -> List[Finding]:
    """Scan Python files for memory/state security patterns (expanded)."""
    findings = []
    try:
        source = path.read_text(encoding="utf-8")
        tree = ast.parse(source, filename=str(path))

        call_nodes = [n for n in ast.walk(tree) if isinstance(n, ast.Call)]

        for node in call_nodes:
            func_name = _get_call_name(node.func)
            if not func_name:
                continue

            func_lower = func_name.lower()
            if any(f in func_lower for f in ["save", "load", "write", "persist", "archive", "cache", "memory"]):
                source_snippet = _get_line_snippet(source, node.lineno)

                for rule in context.rules:
                    for pattern in rule["patterns"]:
                        if re.search(pattern, source_snippet, re.IGNORECASE):
                            # Compute additional metrics
                            state_analysis = analyze_state_persistence(source_snippet)
                            tampering = score_state_tampering(source_snippet, source)
                            overflow = score_context_overflow(source_snippet)
                            compression = score_memory_compression(source_snippet)
                            to_code = score_memory_to_code(source_snippet)
                            exfil = score_memory_exfiltration(source_snippet)
                            versioning = score_memory_versioning(source_snippet)
                            poisoning = score_context_poisoning(source_snippet)
                            secrets = detect_secrets_in_state(source_snippet)
                            framework = detect_framework(path, source_snippet)

                            evidence = rule["evidence_extractor"](node, None) | {
                                "file": str(path),
                                "code_line": source_snippet.strip(),
                                "state_analysis": state_analysis,
                                "tampering": tampering,
                                "context_overflow": overflow,
                                "compression": compression,
                                "to_code": to_code,
                                "exfiltration": exfil,
                                "versioning": versioning,
                                "poisoning": poisoning,
                                "secrets": secrets,
                                "frameworks": framework,
                            }

                            findings.append(
                                Finding(
                                    id=rule["id"],
                                    title=rule["name"],
                                    severity=rule["severity"],
                                    category=rule["category"],
                                    description=f"Found {func_name} usage matching pattern: {pattern}",
                                    evidence=evidence,
                                    location=f"{path}:{node.lineno}",
                                    rule_id=rule["id"],
                                    remediation_hint=rule.get("remediation_hint"),
                                    references=[
                                        "https://langchain-ai.github.io/langchain/",
                                        "https://microsoft.github.io/autogen/",
                                    ],
                                    tags=[
                                        "persistence" if "save" in func_lower else "memory",
                                        "tampering" if tampering["is_vulnerable"] else None,
                                        "secrets" if secrets else None,
                                    ] + framework,
                                    context={
                                        "framework": framework[0] if framework else "generic",
                                        "state_analysis": state_analysis,
                                        "tampering": tampering,
                                        "context_overflow": overflow,
                                        "compression": compression,
                                        "to_code": to_code,
                                        "exfiltration": exfil,
                                        "versioning": versioning,
                                        "poisoning": poisoning,
                                    },
                                    file_line_range=(node.lineno, node.end_lineno or node.lineno),
                                )
                            )
    except Exception as e:
        findings.append(
            Finding(
                id="MEM-SEC-099",
                title="AST parse failed",
                severity=Severity.LOW,
                category=Category.SECRET_EXPOSURE,
                description=f"Failed to parse {path}: {e}",
                evidence={"exception": str(e)},
                context={"path": str(path)},
            )
        )
    return findings



def _get_call_name(func_node: ast.AST) -> Optional[str]:
    if isinstance(func_node, ast.Name):
        return func_node.id
    if isinstance(func_node, ast.Attribute):
        return f"{_get_call_name(func_node.value)}.{func_node.attr}"
    if isinstance(func_node, ast.Call):
        return f"{_get_call_name(func_node.func)}()"
    return None



def _get_line_snippet(source: str, lineno: int, context_lines: int = 3) -> str:
    lines = source.splitlines()
    start = max(0, lineno - 1 - context_lines)
    end = min(len(lines), lineno + context_lines)
    return "\n".join(
        f"{i + 1:>5} | {line}"
        for i, line in enumerate(lines[start:end], start=start)
    )
# === MCP STATE STORAGE PARSER (PART 2) ===

def parse_mcp_state_storage(root: Path) -> List[Dict[str, Any]]:
    """Parse MCP config for state storage patterns (e.g., state stores, context stores)."""
    state_stores = []
    config_files = list(root.rglob("mcp*.yaml")) + list(root.rglob("mcp*.yml")) + list(root.rglob("mcp*.json"))
    for cfg_path in config_files:
        try:
            text = cfg_path.read_text()
            data = json.loads(text) if cfg_path.suffix == ".json" else _load_yaml(text)

            # Look for state stores at top-level or under 'state' / 'memory' keys
            if "state" in data:
                states = data["state"] if isinstance(data["state"], list) else [data["state"]]
                for state in states:
                    if isinstance(state, dict):
                        state_stores.append({
                            "path": state.get("path", "/"),
                            "type": state.get("type", "memory"),
                            "mode": state.get("mode", "read-write"),
                            "source": str(cfg_path),
                        })
            if "memory" in data:
                memories = data["memory"] if isinstance(data["memory"], list) else [data["memory"]]
                for mem in memories:
                    if isinstance(mem, dict):
                        state_stores.append({
                            "path": mem.get("path", "/"),
                            "type": mem.get("type", "chat"),
                            "mode": mem.get("mode", "read-write"),
                            "source": str(cfg_path),
                        })
        except Exception:
            continue
    return state_stores



def _load_yaml(text: str) -> Dict[str, Any]:
    import yaml
    return yaml.safe_load(text) or {}



# === SECURITY SCORING ENGINE (PART 2) ===

def compute_risk_score(finding: Finding, context: ScannerContext = None) -> Dict[str, Any]:
    """Compute dynamic risk score (0–100) and confidence level for memory/state findings."""
    base = {
        "critical": 90,
        "high": 70,
        "medium": 50,
        "low": 30,
    }.get(finding.severity.value, 40)

    # Evidence modifiers
    evidence = finding.evidence
    tampering = evidence.get("tampering", {})
    overflow = evidence.get("context_overflow", {})
    compression = evidence.get("compression", {})
    to_code = evidence.get("to_code", {})
    exfil = evidence.get("exfiltration", {})
    versioning = evidence.get("versioning", {})
    poisoning = evidence.get("poisoning", {})

    tampering_modifier = tampering.get("tamper_risk_score", 0) if isinstance(tampering, dict) else 0
    overflow_modifier = overflow.get("overflow_risk_score", 0) if isinstance(overflow, dict) else 0
    compression_modifier = compression.get("compression_risk_score", 0) if isinstance(compression, dict) else 0
    to_code_modifier = to_code.get("to_code_risk_score", 0) if isinstance(to_code, dict) else 0
    exfil_modifier = exfil.get("exfil_risk_score", 0) if isinstance(exfil, dict) else 0
    versioning_modifier = versioning.get("versioning_risk_score", 0) if isinstance(versioning, dict) else 0
    poisoning_modifier = poisoning.get("poisoning_risk_score", 0) if isinstance(poisoning, dict) else 0

    final_score = min(
        100,
        base
        + tampering_modifier
        + overflow_modifier
        + compression_modifier
        + to_code_modifier
        + exfil_modifier
        + versioning_modifier
        + poisoning_modifier,
    )

    # Confidence: based on evidence richness
    evidence_items = list(evidence.items())
    confidence = 0.5 + (len(evidence_items) / 12) # up to +0.5

    return {
        "risk_score": final_score,
        "tampering_modifier": tampering_modifier,
        "overflow_modifier": overflow_modifier,
        "compression_modifier": compression_modifier,
        "to_code_modifier": to_code_modifier,
        "exfil_modifier": exfil_modifier,
        "versioning_modifier": versioning_modifier,
        "poisoning_modifier": poisoning_modifier,
        "confidence": round(confidence, 2),
    }
