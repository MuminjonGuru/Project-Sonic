# === PATH ENTROPY & DYNAMISM (NEW) ===

def calculate_path_entropy(path: str) -> float:
    """
    Calculate Shannon entropy of a path — higher = more dynamic.
    e.g., 'logs/{user_id}/{date}/output.json' → high entropy
    """
    dynamic_segments = re.findall(r"\$\{.*?\}|\{.*?\}|%\w+%|\$\w+", path)
    if not dynamic_segments:
        return 1.0 # Static path

    entropy = 0.0
    segment_counts: Dict[str, int] = {}

    for seg in dynamic_segments:
        segment_counts[seg] = segment_counts.get(seg, 0) + 1

    total = len(dynamic_segments)
    for count in segment_counts.values():
        p = count / total
        entropy -= p * math.log2(p)

    return entropy



def is_path_dynamic(path: str) -> bool:
    """Return True if path contains dynamic segments."""
    dynamic_patterns = [r"\{.*?\}", r"\$\{.*?\}", r"%\w+%"]
    return any(re.search(p, path) for p in dynamic_patterns)



def calculate_path_volatility(path: str) -> Dict[str, Any]:
    """Calculate path volatility: how likely a path is to change at runtime."""
    parts = path.split("/")
    volatility = 0.0
    for part in parts:
        if re.search(r"\{.*?\}", part):
            volatility += 0.3
        if re.search(r"\$\{.*?\}", part):
            volatility += 0.35
        if re.search(r"\*\*?", part):
            volatility += 0.25
    return {
        "score": min(volatility, 1.0),
        "high_risk_parts": [
            p for p in parts if re.search(r"\{.*?\}|\$\{.*?\}|\*\*?", p)
        ],
    }



# === WRITE DEPTH ANALYSIS (NEW) ===

def analyze_write_depth(path: str, base_paths: List[str] = None) -> Dict[str, Any]:
    """Analyze write depth: how far into the filesystem tree a path goes."""
    if base_paths is None:
        base_paths = ["", "/tmp", "/run", "/home", "/app"]

    path_obj = Path(path)
    parts = [p for p in path_obj.parts if p not in {"", ".", "..", "..\\", "..//"}]

    depth = len(parts)
    is_deep = depth > 4 # arbitrary but reasonable threshold
    risk_score = min(100, depth * 15 + (20 if is_deep else 0))

    safe_base = any(str(path_obj).startswith(bp) for bp in base_paths)
    relative_depth = depth - (1 if safe_base else 0)

    return {
        "depth": depth,
        "is_deep": is_deep,
        "relative_depth": relative_depth,
        "risk_score": risk_score,
    }



# === MCP FILESYSTEM ROOT TRUST MODEL (NEW) ===

def evaluate_mcp_root_trust(root: Dict[str, Any]) -> Dict[str, Any]:
    """Evaluate MCP filesystem root trust."""
    permissions = root.get("permissions", [])
    path = root.get("path", "/")
    is_explicit_write = "write" in permissions
    is_wildcard = "**" in path or "*" in path

    trust_score = 0
    if is_explicit_write:
        trust_score += 40
    if is_wildcard:
        trust_score += 30
    if not is_explicit_write and not is_wildcard:
        trust_score += 20 # implicit trust

    return {
        "explicit_write": is_explicit_write,
        "wildcard_depth": is_wildcard,
        "trust_score": trust_score,
        "risk_level": "high" if trust_score < 40 else "medium" if trust_score < 70 else "low",
    }

def detect_secrets_in_file(file_content: str) -> List[Dict[str, Any]]:
    """Detect unmasked secrets in file content (e.g., logs, configs)."""
    secrets_found = []
    for cat, patterns in SECRET_PATTERNS.items():
        for pattern in patterns:
            matches = re.findall(rf"(?i){pattern}[\s:=]+\s*['\"]?[\w\-\_]+", file_content)
            if matches:
                secrets_found.append({
                    "category": cat,
                    "pattern": pattern,
                    "matches": matches[:3], # sample up to 3 matches
                })
    return secrets_found

# === TRAVERSAL SAFETY SCORER (NEW) ===

def score_traversal_safety(path: str, base_path: str = None) -> Dict[str, Any]:
    """Score how safe a traversal-heavy path is."""
    depth = path.count("..")
    safe_indicators = [
        r"\.resolve\(\)",
        r"os\.path\.commonpath",
        r"Path\(\).*\.resolve",
    ]
    safety_score = 100 - depth * 10
    for pattern in safe_indicators:
        if re.search(pattern, path):
            safety_score += 15
    safety_score = min(100, safety_score)
    return {
        "traversal_depth": depth,
        "safety_score": safety_score,
        "is_safe": safety_score >= 60,
    }



# === FILETYPE RISK SCORER (NEW) ===

def evaluate_filetype_risk(path: Path, content: Optional[str] = None) -> Dict[str, Any]:
    """Evaluate risk profile for a file based on extension and content."""
    ext = path.suffix.lower()
    risk = FILETYPE_RISK.get(ext.lstrip("."), {"extensions": [ext], "risk_reason": "generic"})

    code_like = False
    if content:
        code_like = any(
            p in content for p in ["eval(", "exec(", "import ", "def ", "class ", "return "]
        )

    return {
        "extension": ext,
        "risk_category": risk["risk_reason"],
        "code_like": code_like,
        "risk_score": 80 if ext in [".py"] else 60 if ext in [".yaml", ".yml"] else 40,
    }



# === SECURITY SCORING ENGINE (NEW) ===

def compute_risk_score(finding: Finding, context: ScannerContext = None) -> Dict[str, Any]:
    """Compute dynamic risk score (0–100) and confidence level."""
    base = {
        "critical": 90,
        "high": 70,
        "medium": 50,
        "low": 30,
    }.get(finding.severity.value, 40)

    entropy = finding.evidence.get("path_entropy", 0)
    entropy_modifier = int(entropy * 15) if isinstance(entropy, float) else 0

    depth = finding.evidence.get("write_depth", {})
    depth_modifier = (
        depth.get("risk_score", 0) if isinstance(depth, dict) else depth * 5
    )

    if "agent" in finding.evidence.get("source", ""):
        context_modifier = 10
    elif "llm" in finding.evidence.get("source", ""):
        context_modifier = 15
    else:
        context_modifier = 0

    final_score = min(100, base + entropy_modifier + depth_modifier + context_modifier)

    evidence_items = list(finding.evidence.items())
    confidence = 0.5 + (len(evidence_items) / 8)

    return {
        "risk_score": final_score,
        "entropy_modifier": entropy_modifier,
        "depth_modifier": depth_modifier,
        "context_modifier": context_modifier,
        "confidence": round(confidence, 2),
    }

# === AST SCANNER (FINAL) — START — ===

def scan_ast_file(path: Path, context: ScannerContext) -> List[Finding]:
    """Scan Python files for filesystem access patterns (final)."""
    findings = []
    try:
        source = path.read_text(encoding="utf-8")
        tree = ast.parse(source, filename=str(path))

        call_nodes = [n for n in ast.walk(tree) if isinstance(n, ast.Call)]

        for node in call_nodes:
            func_name = _get_call_name(node.func)
            if not func_name:
                continue

            func_lower = func_name.lower()
            if any(f in func_lower for f in ["open", "read", "write", "copy", "save", "load"]):
                source_snippet = _get_line_snippet(source, node.lineno)

                for rule in context.rules:
                    for pattern in rule["patterns"]:
                        if re.search(pattern, source_snippet, re.IGNORECASE):
                            # Compute additional metrics
                            path_match = re.search(r"(?:Path\(|f['\"].*?)(\$\{.*?\}|\{.*?\}|%.*?%)", source_snippet)
                            entropy = calculate_path_entropy(source_snippet) if path_match else None
                            depth_analysis = analyze_write_depth(source_snippet) if "write" in func_lower else {}

                            evidence = rule["evidence_extractor"](node, None) | {
                                "file": str(path),
                                "code_line": source_snippet.strip(),
                                "path_entropy": entropy,
                                "write_depth": depth_analysis,
                                "path_volatility": calculate_path_volatility(source_snippet) if path_match else {},
                                "filetype_risk": evaluate_filetype_risk(path),
                            }

                            # Adjust severity based on metrics
                            severity = rule["severity"]
                            if entropy and entropy > 3.0:
                                severity = Severity.HIGH
                            if depth_analysis.get("is_deep"):
                                severity = Severity.MEDIUM if severity.value == "low" else severity

                            findings.append(
                                Finding(
                                    id=rule["id"],
                                    title=rule["name"],
                                    severity=severity,
                                    category=rule["category"],
                                    description=f"Found {func_name} usage matching pattern: {pattern}",
                                    evidence=evidence,
                                    location=f"{path}:{node.lineno}",
                                    rule_id=rule["id"],
                                    remediation_hint=rule.get("remediation_hint"),
                                    references=[
                                        "https://owasp.org/www-community/vulnerabilities/Insecure_File_Upload",
                                        "https://cwe.mitre.org/data/definitions/59.html",
                                    ],
                                    tags=[
                                        "dynamic" if "$" in source_snippet else "static",
                                        "write" if "write" in func_lower else "read",
                                        "traversal" if ".." in source_snippet else "safe",
                                        "high-entropy" if entropy and entropy > 2.5 else None,
                                        "deep-write" if depth_analysis.get("is_deep") else None,
                                    ],
                                    context={
                                        "framework": "generic",
                                        "entropy": entropy,
                                        "write_depth": depth_analysis,
                                        "volatility": calculate_path_volatility(source_snippet) if path_match else {},
                                        "filetype_risk": evaluate_filetype_risk(path),
                                    },
                                    file_line_range=(node.lineno, node.end_lineno or node.lineno),
                                )
                            )
    except Exception as e:
        findings.append(
            Finding(
                id="FS-ACCESS-099",
                title="AST parse failed",
                severity=Severity.LOW,
                category=Category.SECRET_HYGIENE,
                description=f"Failed to parse {path}: {e}",
                evidence={"exception": str(e)},
                context={"path": str(path)},
            )
        )
    return findings



def _get_call_name(func_node: ast.AST) -> Optional[str]:
    if isinstance(func_node, ast.Name):
        return func_node.id
    if isinstance(func_node, ast.Attribute):
        return f"{_get_call_name(func_node.value)}.{func_node.attr}"
    if isinstance(func_node, ast.Call):
        return f"{_get_call_name(func_node.func)}()"
    return None



def _get_line_snippet(source: str, lineno: int, context_lines: int = 3) -> str:
    lines = source.splitlines()
    start = max(0, lineno - 1 - context_lines)
    end = min(len(lines), lineno + context_lines)
    return "\n".join(
        f"{i + 1:>5} | {line}"
        for i, line in enumerate(lines[start:end], start=start)
    )



# === MCP FILESYSTEM ROOT PARSER (FINAL) ===

def parse_mcp_filesystem_roots(root: Path) -> List[Dict[str, Any]]:
    """Parse MCP filesystem roots from mcp.yaml/yml/json."""
    roots = []
    config_files = list(root.rglob("mcp*.yaml")) + list(root.rglob("mcp*.yml")) + list(root.rglob("mcp*.json"))
    for cfg_path in config_files:
        try:
            text = cfg_path.read_text()
            data = json.loads(text) if cfg_path.suffix == ".json" else _load_yaml(text)
            if "filesystem" in data or "filesystems" in data:
                fs = data.get("filesystem", data.get("filesystems", []))
                if isinstance(fs, dict):
                    fs = [fs]
                for fr in fs:
                    if isinstance(fr, dict):
                        roots.append({
                            "path": fr.get("path", "/"),
                            "permissions": fr.get("permissions", []),
                            "description": fr.get("description", ""),
                            "source": str(cfg_path),
                        })
        except Exception:
            continue
    return roots



def _load_yaml(text: str) -> Dict[str, Any]:
    import yaml
    return yaml.safe_load(text) or {}
