def is_a2a_operation(func_name: str) -> bool:
    """Detect if function is likely an A2A operation."""
    func_lower = func_name.lower()
    return any(
        p in func_lower
        for p in ["call", "route", "forward", "share", "authorize", "trust", "sign"]
    )



def analyze_a2a_persistence(path: str, content: Optional[str] = None) -> Dict[str, Any]:
    """Analyze A2A persistence: file, db, or cloud."""
    if content and any(p in content for p in ["db.save", "redis.set", "sqlite.execute"]):
        return {"type": "db", "confidence": 0.8}
    if any(path.endswith(ext) for ext in [".json", ".yaml", ".pkl"]):
        return {"type": "file", "confidence": 0.7}
    if any(ext in path.lower() for ext in ["/s3/", "/gcs/", "azure/"]):
        return {"type": "cloud", "confidence": 0.9}
    return {"type": "unknown", "confidence": 0.5}



# === IMPERSONATION VECTORS SCORER (NEW) ===

def score_impersonation_vectors(code_snippet: str) -> Dict[str, Any]:
    """Score how likely code includes impersonation vectors."""
    imp_vectors = 0
    if re.search(r"agent\.update\s*\(\s*prompt\s*=\s*f['\"].*I am agent", code_snippet):
        imp_vectors += 25
    if re.search(r"agent\.rewrite_system_prompt\s*\(\s*agent\s*=\s*agent2", code_snippet):
        imp_vectors += 25
    if re.search(r"agent\.sign\s*\(\s*agent2", code_snippet):
        imp_vectors += 25
    if re.search(r"agent\.impersonate\s*\(\s*agent\s*=\s*agent2", code_snippet):
        imp_vectors += 25

    return {
        "imp_vectors_score": min(100, imp_vectors),
        "imp_vectors": [
            "prompt-injection",
            "system-prompt-rewrite",
            "signature-based",
            "impersonate-agent",
        ],
        "is_vulnerable": imp_vectors >= 50,
    }



# === TRUST CHAIN SCORER (ENHANCED) ===

def score_trust_chains(code_snippet: str) -> Dict[str, Any]:
    """Score how likely code includes trust chain patterns."""
    trust_risks = 0
    if re.search(r"agent1\.authorize\s*\(\s*agent2", code_snippet):
        trust_risks += 25
    if re.search(r"agent2\.trust\s*\(\s*agent3", code_snippet):
        trust_risks += 25
    if re.search(r"agent\.trust_chain\s*\(\s*agent\s*=\s*agent2", code_snippet):
        trust_risks += 25
    if re.search(r"agent\.verify\s*\(\s*agent2", code_snippet):
        trust_risks += 25

    return {
        "trust_risk_score": min(100, trust_risks),
        "trust_vectors": [
            "authorize",
            "trust",
            "trust_chain",
            "verify",
        ],
        "is_trust_chain": trust_risks >= 50,
    }



# === CROSS-AGENT STATE SCORER (ENHANCED) ===

def score_cross_agent_state(code_snippet: str) -> Dict[str, Any]:
    """Score how likely code includes cross-agent state/memory sharing."""
    state_risks = 0
    if re.search(r"agent1\.memory\s*→\s*agent2\.context", code_snippet):
        state_risks += 25
    if re.search(r"agent1\.state\s*→\s*agent2\.input", code_snippet):
        state_risks += 25
    if re.search(r"agent\.share_memory\s*\(\s*to\s*=\s*agent2", code_snippet):
        state_risks += 25
    if re.search(r"agent\.forward.*agent2", code_snippet):
        state_risks += 25

    return {
        "state_risk_score": min(100, state_risks),
        "state_vectors": [
            "memory-forward",
            "state-forward",
            "share_memory",
            "forward-agent",
        ],
        "is_vulnerable": state_risks >= 50,
    }

FRAMEWORK_PATTERNS = {
    "langchain": [
        r"AgentExecutor",
        r"SequentialChain",
        r"AgentRunner",
    ],
    "autogen": [
        r"GroupChat",
        r"ManagerAgent",
        r"ConversableAgent\.group_chat",
    ],
    "mcp": [
        r"state\.store.*agent",
        r"memory\.persist.*agent",
        r"trust\.store.*agent",
    ],
}



def detect_framework(path: Path, source: str) -> List[str]:
    """Detect which agent frameworks are used in the code."""
    frameworks = []
    for fw, patterns in FRAMEWORK_PATTERNS.items():
        if any(re.search(p, source, re.IGNORECASE) for p in patterns):
            frameworks.append(fw)
    return frameworks



def score_framework_mismatch(code_snippet: str, frameworks: List[str]) -> Dict[str, Any]:
    """Score how likely framework-specific patterns are mismatched."""
    mismatch_risks = 0
    if "langchain" in frameworks and re.search(r"AgentExecutor", code_snippet):
        mismatch_risks += 20
    if "autogen" in frameworks and re.search(r"GroupChat", code_snippet):
        mismatch_risks += 20
    if "mcp" in frameworks and re.search(r"state\.store.*agent", code_snippet):
        mismatch_risks += 20
    if len(frameworks) > 2:
        mismatch_risks += 20 # multi-framework = higher mismatch risk

    return {
        "mismatch_risk_score": min(100, mismatch_risks),
        "mismatch_vectors": frameworks,
        "is_vulnerable": mismatch_risks >= 40,
    }

def scan_ast_file(path: Path, context: ScannerContext) -> List[Finding]:
    """Scan Python files for agent-to-agent patterns (expanded)."""
    findings = []
    try:
        source = path.read_text(encoding="utf-8")
        tree = ast.parse(source, filename=str(path))

        call_nodes = [n for n in ast.walk(tree) if isinstance(n, ast.Call)]

        for node in call_nodes:
            func_name = _get_call_name(node.func)
            if not func_name:
                continue

            func_lower = func_name.lower()
            if any(f in func_lower for f in ["call", "route", "forward", "share", "authorize", "trust", "sign"]):
                source_snippet = _get_line_snippet(source, node.lineno)

                for rule in context.rules:
                    for pattern in rule["patterns"]:
                        if re.search(pattern, source_snippet, re.IGNORECASE):
                            # Compute additional metrics
                            impersonation = score_impersonation_vectors(source_snippet)
                            trust = score_trust_chains(source_snippet)
                            state = score_cross_agent_state(source_snippet)
                            framework = detect_framework(path, source_snippet)
                            framework_mismatch = score_framework_mismatch(source_snippet, framework)

                            evidence = rule["evidence_extractor"](node, None) | {
                                "file": str(path),
                                "code_line": source_snippet.strip(),
                                "impersonation": impersonation,
                                "trust": trust,
                                "state": state,
                                "framework_mismatch": framework_mismatch,
                                "frameworks": framework,
                            }

                            findings.append(
                                Finding(
                                    id=rule["id"],
                                    title=rule["name"],
                                    severity=rule["severity"],
                                    category=rule["category"],
                                    description=f"Found {func_name} usage matching pattern: {pattern}",
                                    evidence=evidence,
                                    location=f"{path}:{node.lineno}",
                                    rule_id=rule["id"],
                                    remediation_hint=rule.get("remediation_hint"),
                                    references=[
                                        "https://langchain-ai.github.io/langchain/",
                                        "https://microsoft.github.io/autogen/",
                                    ],
                                    tags=[
                                        "a2a" if "call" in func_lower else "routing",
                                        "impersonation" if impersonation["is_vulnerable"] else None,
                                        "trust" if trust["is_trust_chain"] else None,
                                        "state" if state["is_vulnerable"] else None,
                                    ] + framework,
                                    context={
                                        "framework": framework[0] if framework else "generic",
                                        "impersonation": impersonation,
                                        "trust": trust,
                                        "state": state,
                                        "framework_mismatch": framework_mismatch,
                                    },
                                    file_line_range=(node.lineno, node.end_lineno or node.lineno),
                                )
                            )
    except Exception as e:
        findings.append(
            Finding(
                id="A2A-099",
                title="AST parse failed",
                severity=Severity.LOW,
                category=Category.A2A_CALL,
                description=f"Failed to parse {path}: {e}",
                evidence={"exception": str(e)},
                context={"path": str(path)},
            )
        )
    return findings



def _get_call_name(func_node: ast.AST) -> Optional[str]:
    if isinstance(func_node, ast.Name):
        return func_node.id
    if isinstance(func_node, ast.Attribute):
        return f"{_get_call_name(func_node.value)}.{func_node.attr}"
    if isinstance(func_node, ast.Call):
        return f"{_get_call_name(func_node.func)}()"
    return None



def _get_line_snippet(source: str, lineno: int, context_lines: int = 3) -> str:
    lines = source.splitlines()
    start = max(0, lineno - 1 - context_lines)
    end = min(len(lines), lineno + context_lines)
    return "\n".join(
        f"{i + 1:>5} | {line}"
        for i, line in enumerate(lines[start:end], start=start)
    )
# === NESTED A2A DETECTOR (NEW) ===

def score_nested_a2a(code_snippet: str) -> Dict[str, Any]:
    """Score how likely code includes nested A2A patterns."""
    nested_risks = 0
    if re.search(r"agent\.call\(agent2\.call\(agent3\)\)", code_snippet):
        nested_risks += 30
    if re.search(r"agent\.route\(task\s*=\s*agent2\.response\)", code_snippet):
        nested_risks += 25
    if re.search(r"agent\.forward\(data\s*=\s*agent2\.call\(agent3\)\)", code_snippet):
        nested_risks += 25
    if re.search(r"agent\.share_memory\(to\s*=\s*agent2\.get_state\(\)\)", code_snippet):
        nested_risks += 20

    return {
        "nested_risk_score": min(100, nested_risks),
        "nested_vectors": [
            "nested-calls",
            "cross-agent-calls",
            "response-forward",
            "state-get",
        ],
        "is_nested": nested_risks >= 60,
    }



# === AUTHORIZATION DRIFT DETECTOR (NEW) ===

def score_authz_drift(code_snippet: str) -> Dict[str, Any]:
    """Score how likely code includes authorization drift patterns."""
    drift_risks = 0
    if re.search(r"agent\.role\s*=\s*agent2\.role", code_snippet):
        drift_risks += 25
    if re.search(r"agent\.permissions\s*=\s*agent2\.permissions", code_snippet):
        drift_risks += 25
    if re.search(r"agent\.update_authorization\s*\(\s*agent\s*=\s*agent2", code_snippet):
        drift_risks += 25
    if re.search(r"agent\.reauthorize\s*\(\s*agent\s*=\s*agent2", code_snippet):
        drift_risks += 25

    return {
        "drift_risk_score": min(100, drift_risks),
        "drift_vectors": [
            "role-drift",
            "permission-drift",
            "update_authorization",
            "reauthorize",
        ],
        "is_vulnerable": drift_risks >= 50,
    }



# === SECURITY SCORING ENGINE (ENHANCED) ===

def compute_risk_score(finding: Finding, context: ScannerContext = None) -> Dict[str, Any]:
    """Compute dynamic risk score (0–100) and confidence level for A2A findings."""
    base = {
        "critical": 90,
        "high": 70,
        "medium": 50,
        "low": 30,
    }.get(finding.severity.value, 40)

    evidence = finding.evidence
    nested = evidence.get("nested", {})
    impersonation = evidence.get("impersonation", {})
    trust = evidence.get("trust", {})
    state = evidence.get("state", {})
    authz_drift = evidence.get("authz_drift", {})
    framework_mismatch = evidence.get("framework_mismatch", {})

    nested_modifier = nested.get("nested_risk_score", 0) if isinstance(nested, dict) else 0
    impersonation_modifier = impersonation.get("imp_vectors_score", 0) if isinstance(impersonation, dict) else 0
    trust_modifier = trust.get("trust_risk_score", 0) if isinstance(trust, dict) else 0
    state_modifier = state.get("state_risk_score", 0) if isinstance(state, dict) else 0
    authz_modifier = authz_drift.get("drift_risk_score", 0) if isinstance(authz_drift, dict) else 0
    framework_modifier = framework_mismatch.get("mismatch_risk_score", 0) if isinstance(framework_mismatch, dict) else 0

    final_score = min(
        100,
        base
        + nested_modifier
        + impersonation_modifier
        + trust_modifier
        + state_modifier
        + authz_modifier
        + framework_modifier,
    )

    evidence_items = list(evidence.items())
    confidence = 0.5 + (len(evidence_items) / 10)

    return {
        "risk_score": final_score,
        "nested_modifier": nested_modifier,
        "impersonation_modifier": impersonation_modifier,
        "trust_modifier": trust_modifier,
        "state_modifier": state_modifier,
        "authz_modifier": authz_modifier,
        "framework_modifier": framework_modifier,
        "confidence": round(confidence, 2),
    }

# === MCP A2A STORAGE PARSER (PART 3) ===

def parse_mcp_a2a_storage(root: Path) -> List[Dict[str, Any]]:
    """Parse MCP config for A2A storage patterns (e.g., trust stores, state_sharing)."""
    state_stores = []
    config_files = list(root.rglob("mcp*.yaml")) + list(root.rglob("mcp*.yml")) + list(root.rglob("mcp*.json"))
    for cfg_path in config_files:
        try:
            text = cfg_path.read_text()
            data = json.loads(text) if cfg_path.suffix == ".json" else _load_yaml(text)

            # Look for A2A storage at top-level or under 'trust', 'state_sharing' keys
            if "trust" in data:
                trusts = data["trust"] if isinstance(data["trust"], list) else [data["trust"]]
                for trust in trusts:
                    if isinstance(trust, dict):
                        state_stores.append({
                            "path": trust.get("path", "/"),
                            "type": "trust",
                            "mode": trust.get("mode", "read-write"),
                            "source": str(cfg_path),
                        })
            if "state_sharing" in data:
                sharings = data["state_sharing"] if isinstance(data["state_sharing"], list) else [data["state_sharing"]]
                for sharing in sharings:
                    if isinstance(sharing, dict):
                        state_stores.append({
                            "path": sharing.get("path", "/"),
                            "type": "state_sharing",
                            "mode": sharing.get("mode", "read-write"),
                            "source": str(cfg_path),
                        })
        except Exception:
            continue
    return state_stores



def _load_yaml(text: str) -> Dict[str, Any]:
    import yaml
    return yaml.safe_load(text) or {}
